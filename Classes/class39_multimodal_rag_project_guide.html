## src/parsing.py — ComplexPDFParser class skeleton

import os, json, logging
import fitz                             # PyMuPDF — text + image extraction
import camelot                          # table extraction from PDFs
import pytesseract                      # OCR for scanned images
from PIL import Image
from langchain.schema import Document
from pathlib import Path
from logger.custom_logger import setup_logger
from exception.custom_exception import DocumentPortalException

logger = setup_logger(log_file="logs/parsing.log")

class ComplexPDFParser:
    """Parses complex PDFs: extracts text, images, and tables."""

    def __init__(self, pdf_path: str, output_dir: str = "./data/parsed"):
        self.pdf_path        = pdf_path
        self.output_dir      = output_dir
        self.image_dir       = os.path.join(output_dir, "images")
        self.page_image_dir  = os.path.join(output_dir, "page_images")
        self.render_scale    = 2.0         # rendering DPI multiplier
        self.page_record     = []            # text content per page
        self.image_record    = []            # embedded images with OCR text
        self.table_record    = []            # extracted table data
        self.langchain_docs  = []            # LangChain Document objects
        logger.info("Parser initialised", pdf=pdf_path)

    def _validate_pdf(self) -> None:
        """Check the PDF file exists and is a valid PDF."""
        if not os.path.exists(self.pdf_path):
            raise DocumentPortalException(ff"PDF not found: {self.pdf_path}")
        if not self.pdf_path.lower().endswith(".pdf"):
            raise DocumentPortalException("File must be a .pdf")
        logger.info("PDF validated")

    def _create_output_directories(self) -> None:
        for dir_path in [self.output_dir, self.image_dir, self.page_image_dir]:
            Path(dir_path).mkdir(parents=True, exist_ok=True)

    def _configure_tesseract(self) -> None:
        tesseract_path = os.getenv("TESSERACT_PATH", "/usr/bin/tesseract")
        pytesseract.pytesseract.tesseract_cmd = tesseract_path

    def _run_ocr_on_images(self, image_path: str) -> str:
        """Run Tesseract OCR on an image file → return extracted text."""
        try:
            img = Image.open(image_path)
            return pytesseract.image_to_string(img)
        except Exception as e:
            logger.warning("OCR failed", image=image_path, error=str(e))
            return ""

    def extract_text_and_images(self) -> None:
        """Extract native PDF text and render pages as images."""
        doc = fitz.open(self.pdf_path)
        for page_num, page in enumerate(doc):
            text = page.get_text()
            # Render page as high-res image
            mat    = fitz.Matrix(self.render_scale, self.render_scale)
            pixmap = page.get_pixmap(matrix=mat)
            img_path = os.path.join(self.page_image_dir, ff"page_{page_num+1}.png")
            pixmap.save(img_path)
            # OCR the rendered page image for scanned PDFs
            ocr_text = self._run_ocr_on_images(img_path)
            self.page_record.append({
                "page_number": page_num + 1,
                "native_text": text,
                "ocr_text": ocr_text,
                "page_image": img_path
            })
        doc.close()
        logger.info("Text and images extracted", pages=len(self.page_record))

    def extract_table(self) -> None:
        """Extract tables using camelot."""
        try:
            tables = camelot.read_pdf(self.pdf_path, pages="all", flavor="stream")
            for i, table in enumerate(tables):
                self.table_record.append({
                    "table_index": i,
                    "page":        table.page,
                    "data":        table.df.to_dict()
                })
            logger.info("Tables extracted", count=len(self.table_record))
        except Exception as e:
            logger.warning("Table extraction failed", error=str(e))

    def create_langchain_document(self) -> None:
        """Convert page_record into LangChain Document objects."""
        for page in self.page_record:
            content = page["native_text"] or page["ocr_text"]
            if content.strip():
                self.langchain_docs.append(Document(
                    page_content = content,
                    metadata     = {
                        "source":      os.path.basename(self.pdf_path),
                        "page_number": page["page_number"],
                        "page_image":  page["page_image"]
                    }
                ))

    def _save_json(self, data: list, filename: str) -> None:
        path = os.path.join(self.output_dir, filename)
        with open(path, "w") as f:
            json.dump(data, f, indent=2, default=str)

    def save_output(self) -> None:
        self._save_json(self.page_record,  "page_record.json")
        self._save_json(self.image_record, "image_record.json")
        self._save_json(self.table_record, "table_record.json")
        logger.info("Output saved to disk")

    def parse(self) -> dict:
        """Master method — orchestrates the full parsing pipeline."""
        try:
            logger.info("Starting PDF parse", pdf=self.pdf_path)
            self._validate_pdf()
            self._create_output_directories()
            self._configure_tesseract()
            self.extract_text_and_images()
            self.extract_table()
            self.create_langchain_document()
            self.save_output()
            logger.info("Parse complete",
                         pages=len(self.page_record),
                         tables=len(self.table_record),
                         docs=len(self.langchain_docs))
            return {
                "langchain_documents": self.langchain_docs,
                "page_record":         self.page_record,
                "table_record":        self.table_record,
                "image_record":        self.image_record
            }
        except Exception as e:
            logger.error("Parse failed", error=str(e))
            raise DocumentPortalException(ff"Parsing failed: {e}")

## src/ingestion.py — DocumentIngestion class (skeleton)

import os, hashlib
from langchain_openai import OpenAIEmbeddings
from langchain.text_splitter import RecursiveCharacterTextSplitter
from langchain_qdrant import QdrantVectorStore
from qdrant_client import QdrantClient
from qdrant_client.models import Distance, VectorParams
from src.parsing import ComplexPDFParser
from logger.custom_logger import setup_logger
from exception.custom_exception import DocumentPortalException

logger = setup_logger(log_file="logs/ingestion.log")

class DocumentIngestion:
    def __init__(
        self,
        collection_name:     str   = None,
        embedding_model:     str   = "text-embedding-3-small",
        embedding_dimension: int   = 1536,
        batch_size:          int   = 64
    ):
        self.collection_name    = collection_name or os.getenv("QDRANT_COLLECTION_NAME", "rag_documents")
        self.embedding_model    = embedding_model
        self.embedding_dimension= embedding_dimension
        self.batch_size         = batch_size
        self._validate_configuration()

        self.text_splitter = RecursiveCharacterTextSplitter(
            chunk_size    = 1000,
            chunk_overlap = 200
        )
        self.embedding = OpenAIEmbeddings(model=self.embedding_model)
        self.client    = QdrantClient(
            url    = os.getenv("QDRANT_CLUSTER_ENDPOINT"),
            api_key= os.getenv("QDRANT_API_KEY")
        )
        self._ensure_collection()
        self.vector_store = QdrantVectorStore(
            client          = self.client,
            collection_name = self.collection_name,
            embedding       = self.embedding
        )
        logger.info("Ingestion class initialised", collection=self.collection_name)

    def _ensure_collection(self):
        existing = [c.name for c in self.client.get_collections().collections]
        if self.collection_name not in existing:
            self.client.create_collection(
                collection_name = self.collection_name,
                vectors_config  = VectorParams(size=self.embedding_dimension, distance=Distance.COSINE)
            )
            logger.info("Collection created", name=self.collection_name)

    def ingest_pdf(self, pdf_path: str) -> dict:
        """Master method: parse → split → embed → ingest to Qdrant."""
        try:
            logger.info("Starting PDF ingestion", pdf=pdf_path)
            # 1. Parse
            parser       = ComplexPDFParser(pdf_path=pdf_path)
            parsed       = parser.parse()
            documents    = parsed["langchain_documents"]
            # 2. Split (chunk)
            chunks       = self.text_splitter.split_documents(documents)
            logger.info("Chunks created", count=len(chunks))
            # 3. Embed + store in Qdrant
            self.vector_store.add_documents(chunks, batch_size=self.batch_size)
            logger.info("Ingestion complete", chunks_stored=len(chunks))
            return {"status": "success", "chunks_stored": len(chunks)}
        except Exception as e:
            logger.error("Ingestion failed", error=str(e))
            raise DocumentPortalException(ff"Ingestion failed: {e}")

class MultimodalQdrantRetriever:
    def __init__(self, collection_name: str = None, ...):
        # Same Qdrant client setup as Ingestion
        # Load vector store (not create — it already exists)
        self.vector_store = QdrantVectorStore(
            client=self.client,
            collection_name=self.collection_name,
            embedding=self.embedding
        )

    def _build_filter(self, query: str) -> None:
        # Optional: build metadata filter from query context
        return None

    def retrieve(self, query: str, k: int = 5) -> list:
        """Retrieve top-k relevant documents from Qdrant."""
        query_filter = self._build_filter(query)
        results      = self.vector_store.similarity_search(query, k=k)
        logger.info("Retrieval complete", query=query[:50], results=len(results))
        return results

    def retrieve_with_score(self, query: str, k: int = 5) -> list:
        """Retrieve with similarity scores for debugging / thresholding."""
        results = self.vector_store.similarity_search_with_score(query, k=k)
        for doc, score in results:
            logger.debug("Retrieved doc", score=round(score, 4), page=doc.metadata.get("page_number"))
        return results

    def as_langchain_retriever(self, search_type="similarity", k=5):
        """Return a LangChain-compatible retriever object."""
        search_kwargs = {"k": k}
        # Add MMR params if search_type == "mmr"
        if search_type == "mmr":
            search_kwargs.update({"fetch_k": 20, "lambda_mult": 0.5})
        return self.vector_store.as_retriever(
            search_type   = search_type,
            search_kwargs = search_kwargs
        )

## src/generation.py — MultimodalRAGGenerator class

import base64
from langchain_openai import ChatOpenAI
from prompt_library.prompt import (
    MULTIMODAL_RAG_SYSTEM_PROMPT, MULTIMODAL_RAG_USER_PROMPT,
    format_multimodal_rag_user_prompt, format_retrieve_image_evidence_prompt
)
from src.retrieval import MultimodalQdrantRetriever

class MultimodalRAGGenerator:
    def __init__(
        self,
        collection_name: str,
        model_name:      str   = "gpt-4.1-mini",
        temperature:     float = 0.2,
        max_images:      int   = 3,
        max_context_chars:int  = 10000
    ):
        self.max_images       = max_images
        self.max_context_chars= max_context_chars
        self.retriever        = MultimodalQdrantRetriever(collection_name=collection_name)
        self.llm              = ChatOpenAI(model=model_name, temperature=temperature)

    def _image_to_data_url(self, image_path: str) -> str:
        """Convert image file → base64 data URL for multimodal LLM input."""
        with open(image_path, "rb") as f:
            b64 = base64.b64encode(f.read()).decode("utf-8")
        return ff"data:image/png;base64,{b64}"

    def _prepare_context(self, retrieved_docs: list) -> str:
        """Combine retrieved text chunks into a single context string."""
        context_parts = []
        for i, doc in enumerate(retrieved_docs):
            context_parts.append(
                ff"[Source {i+1} | Page {doc.metadata.get('page_number','?')}]\n{doc.page_content}"
            )
        context = "\n\n".join(context_parts)
        return context[:self.max_context_chars]  # truncate if too long

    def _build_messages(self, query: str, context: str, image_urls: list) -> list:
        """Build the final message list to send to the multimodal LLM."""
        # System message
        messages = [("system", MULTIMODAL_RAG_SYSTEM_PROMPT)]
        # User message with context
        user_content = format_multimodal_rag_user_prompt(query=query, context=context)
        # Append images if available
        if image_urls:
            messages.append(("human", [
                {"type": "text", "text": user_content},
                *[{"type": "image_url", "image_url": {"url": url}} for url in image_urls]
            ]))
        else:
            messages.append(("human", user_content))
        return messages

    def generate(self, query: str, k: int = 5) -> dict:
        """Master method: retrieve → build context → call LLM → return answer."""
        try:
            # 1. Retrieve relevant documents
            docs_with_scores = self.retriever.retrieve_with_score(query, k=k)
            docs             = [doc for doc, _ in docs_with_scores]
            # 2. Prepare text context
            context    = self._prepare_context(docs)
            # 3. Collect page images for multimodal input
            image_urls = []
            for doc in docs[:self.max_images]:
                img_path = doc.metadata.get("page_image")
                if img_path and os.path.exists(img_path):
                    image_urls.append(self._image_to_data_url(img_path))
            # 4. Build messages + call LLM
            messages = self._build_messages(query, context, image_urls)
            response = self.llm.invoke(messages)
            return {"answer": response.content, "sources": docs}
        except Exception as e:
            logger.error("Generation failed", error=str(e))
            raise DocumentPortalException(ff"Generation failed: {e}")

# ── System prompt: defines the assistant's role and rules ─────────
MULTIMODAL_RAG_SYSTEM_PROMPT = """You are an answer generator component of a Multimodal RAG system.
Answer ONLY from the retrieved context and retrieved images provided.
Do NOT answer from outside knowledge.
If the retrieved evidence is insufficient, clearly say 'Evidence is not available.'
Treat text, OCR output, table data, and images as equal evidence sources.
If sources disagree, mention the discrepancy.
Be concise, accurate, and cite the source page number where possible."""

{context}

User Question: {query}

Answer the user question ONLY based on the evidence above."""

def setup_logger(log_file: str = None) -> logging.Logger:
    """Create a logger that writes to both console and timestamped log file."""
    # Create logs directory
    log_dir = "logs"
    Path(log_dir).mkdir(exist_ok=True)

    # Timestamped filename: logs/08_17_2026_14_35_22.log
    timestamp = datetime.now().strftime("%m_%d_%Y_%H_%M_%S")
    log_file  = log_file or os.path.join(log_dir, ff"{timestamp}.log")

    logging.basicConfig(
        level   = logging.INFO,
        format  = "%(asctime)s | %(levelname)s | %(name)s | %(message)s",
        handlers= [
            logging.FileHandler(log_file),
            logging.StreamHandler()     # also print to console/terminal
        ]
    )
    return logging.getLogger("MultimodalRAG")

14 — Design Pattern

Decoupled Pipeline Design

"These two things are decoupled, actually, from the entire pipeline — this ingestion and the parsing one. So this thing is a decoupled one. Ingestion and parsing, we can do before-hand, and then on top of it, we can perform our generation, means our chatting. Always keep this ingestion-parsing pipeline as a decoupled one. Schedule it separately. Parsing should be a separate one, ingestion should be a separate one. And generation/retrieval should be the separate pipeline."
Two Separate Pipelines
Pipeline A — Offline (run once / scheduled)
Parse PDF → Chunk → Embed → Store in Qdrant
Run via Airflow, a cron job, or triggered by UI "Index" button. Not in the critical path.
Pipeline B — Online (every user query)
User Query → Retrieve from Qdrant → Generate Answer
Must be fast — only retrieval + generation. Parsing/ingestion already done.

Quiz — Test Your Understanding

📝 Class 39 Concept Check

1. Why can't a Jupyter Notebook be deployed directly to production? What architectural changes are needed?

▶ Show Answer
Notebooks are interactive, sequential, and not structured for deployment. They have no error handling, no logging, no modularity, and can't be served as a web API or scheduled job. To productionize: convert notebook cells into Python classes and methods (OOP); add custom logger for observability; add custom exceptions for robust error handling; break into modules (parsing.py, ingestion.py, retrieval.py, generation.py); create a config system (.env + config.yaml); build a UI (Streamlit/FastAPI); add a setup.py for proper package imports. The code logic is exactly the same — only the structure changes from linear cells to callable classes.

2. What are the 4 pipeline stages and their corresponding files in the Multimodal RAG project?

▶ Show Answer
Stage 1 — Parsing (parsing.py): PDF → text (PyMuPDF) + images (Tesseract OCR) + tables (camelot) → LangChain Documents. Class: ComplexPDFParser. Stage 2 — Ingestion (ingestion.py): LangChain Documents → split into chunks → embed with OpenAI text-embedding-3-small → store in Qdrant vector DB. Class: DocumentIngestion. Stage 3 — Retrieval (retrieval.py): user query → vector similarity search in Qdrant → top-k chunks with scores. Class: MultimodalQdrantRetriever. Stage 4 — Generation (generation.py): retrieved chunks + images + prompt → GPT-4.1-mini → grounded answer. Class: MultimodalRAGGenerator.

3. What is the private method naming convention in Python and what methods in ComplexPDFParser are private vs public?

▶ Show Answer
In Python, methods starting with a single underscore (_methodname) are "private by convention" — they should not be called from outside the class. They're meant to be called only from within the class itself. Private methods in ComplexPDFParser: _validate_pdf(), _create_output_directories(), _configure_tesseract(), _run_ocr_on_images(), _save_json(). Public methods (callable from outside): extract_text_and_images(), extract_table(), create_langchain_document(), save_output(), parse() (master). The pattern: internal helpers are private; the main orchestration method parse() is public and calls all the private helpers internally.

4. What does "decoupled pipeline" mean in the context of this project and why is it important?

▶ Show Answer
Decoupled pipeline means: Pipeline A (parsing + ingestion) and Pipeline B (retrieval + generation) are independent. Pipeline A runs once or on schedule (when new documents arrive). Pipeline B runs on every user query. Why important: (1) Performance — user queries are not slowed down by PDF parsing; only fast retrieval + generation is in the critical path. (2) Scalability — pipelines can scale independently. (3) Reliability — a parsing error doesn't affect user queries if data is already indexed. (4) Scheduling — Pipeline A can be triggered by document upload or run as a nightly batch job. In the Streamlit UI: left sidebar (upload + parse + index) is Pipeline A; chat panel is Pipeline B. Sunny: "Always keep ingestion-parsing as a decoupled one. Schedule it separately."
